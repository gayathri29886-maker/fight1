Reached Level {self.level}! Bonus Awarded: {self.format_currency(bonus)}") # --- GAME ACTIONS --- def expand_land(self): cost = int(self.land_cost * max(0.3, (1 - (self.executive_stacks['eng'] * 0.07)))) if self.money >= cost: self.money -= cost self.land += 1 self.land_cost = int(self.land_cost * 1.05 / 100) * 100 print(f"🏗️ Floor capacity expanded! Total capacity: {self.land} floors.") else: print("❌ Insufficient funds to expand floor capacity!") def build_unit(self, unit_key): if unit_key not in self.unit_types: print("❌ Invalid building type.") return unit = self.unit_types[unit_key] if unit.get('ap') and not self.ap_unlocked: print("🔒 AP Supreme Building locked! Enter operator password first.") return cost = int(unit['cost'] * max(0.2, (1 - (self.executive_stacks['vp'] * 0.06) - (self.executive_stacks['cpo'] * 0.08)))) if self.get_used_plots() + unit['plots'] > self.land: print(f"❌ Not enough floor capacity! Requires {unit['plots']} slots. Expand land first.") return if self.money >= cost: self.money -= cost self.floors.append(unit_key) xp_gain = unit['xp'] * (1 + (self.amenity_stacks['fiber'] * 0.25) + (self.executive_stacks['cio'] * 0.20) + (self.executive_stacks['cvo'] * 0.50)) self.xp += int(xp_gain) self.check_level_up() print(f"✅ Successfully constructed {unit['name']}!") else: print("❌ Insufficient funds for construction!") def upgrade_amenity(self, key): if self.amenity_stacks[key] >= 10: print("⭐ Amenity already maxed out at Level 10!") return base_cost = 15000 * (self.amenity_stacks[key] + 1) cost = int(base_cost * max(0.4, (1 - (self.executive_stacks['chro'] * 0.05)))) if self.money >= cost: self.money -= cost self.amenity_stacks[key] += 1 print(f"✨ Upgraded amenity '{key}' to Level {self.amenity_stacks[key]}/10!") else: print("❌ Insufficient funds for amenity upgrade!") def hire_executive(self, key): if self.executive_stacks[key] >= 10: print("⭐ Executive already maxed out at Level 10!") return base_cost = 30000 * (self.executive_stacks[key] + 1) cost = int(base_cost * max(0.4, (1 - (self.executive_stacks['chro'] * 0.05)))) if self.money >= cost: self.money -= cost self.executive_stacks[key] += 1 print(f"👔 Hired/Upgraded executive '{key}' to Level {self.executive_stacks[key]}/10!") else: print("❌ Insufficient funds to hire/upgrade executive!") def take_loan(self, amount, rate): if self.loan_balance > 0: print("❌ You already have an active bank loan!") return self.loan_balance = amount self.loan_interest_rate = rate self.money += amount print(f"🏦 Secured credit facility of {self.format_currency(amount)} at {rate*100}% interest rate.") def repay_loan(self): if self.loan_balance <= 0: print("ℹ️ No active loan to repay.") return if self.money >= self.loan_balance: self.money -= self.loan_balance print(f"✅ Successfully repaid total loan of {self.format_currency(self.loan_balance)}!") self.loan_balance = 0 self.ticks_since_interest = 0 else: print(f"❌ Insufficient funds! You need {self.format_currency(self.loan_balance)} to clear the loan.") def authenticate_ap(self, password): if password == "AP is OP": self.ap_unlocked = True print("⚡ AP Master Access Granted! Supreme Operator Controls Unlocked.") return True print("❌ Access Denied: Invalid Security Password!") return False def trigger_ap_grant(self, cash=1000000000, floors=50): if not self.ap_unlocked: print("🔒 AP Panel is locked.") return self.money += cash self.land += floors print(f"⚡ AP God Grant: Added {self.format_currency(cash)} & +{floors} Floors!") # --- BACKGROUND SIMULATION TICK ENGINE --- def game_loop(self): while self.running: time.sleep(1.0) # Event Timer if self.event_timer > 0: self.event_timer -= 1 if self.event_timer <= 0: self.current_event = {'id': 'NORMAL', 'name': 'Normal Market Conditions', 'desc': 'Standard climate.', 'rent_mult': 1.0, 'occ_boost': 0.0} elif random.random() < 0.04: events = [ {'id': 'BOOM', 'name': '🌆 Economic Boom', 'desc': 'Tenant demand soars!', 'rent_mult': 1.5, 'occ_boost': 0.3}, {'id': 'TECH', 'name': '💻 Tech Surge', 'desc': 'Server demand spikes!', 'rent_mult': 2.0, 'occ_boost': 0.15}, {'id': 'CRASH', 'name': '📉 Market Crash', 'desc': 'Economic downturn!', 'rent_mult': 0.7, 'occ_boost': -0.2} ] ev = random.choice(events) self.current_event = ev self.event_timer = 30 # Occupancy Calculation base_occ = 0.70 + random.uniform(0.0, 0.45) + self.current_event['occ_boost'] base_occ += (self.amenity_stacks['pool'] * 0.08) + (self.amenity_stacks['spa'] * 0.10) + (self.executive_stacks['cmo'] * 0.08) self.occupancy_factor = min(1.50, max(0.70, base_occ)) # Multipliers total_mult = 1.0 + (self.amenity_stacks['elevators'] * 0.10) + (self.amenity_stacks['solar'] * 0.15) + (self.executive_stacks['strategy'] * 0.15) total_rent = int(total_rent * total_mult) if self.stocks['grit'] > 0: total_rent += int(self.stocks['grit'] * 80 * (1 + (self.executive_stacks['ir'] * 0.25))) # Loan Interest if self.loan_balance > 0: self.ticks_since_interest += 1 if self.ticks_since_interest >= 10: self.ticks_since_interest = 0 rate = self.loan_interest_rate * max(0.1, (1 - (self.executive_stacks['cfo'] * 0.10))) self.loan_balance += int(self.loan_balance * rate) self.money += total_rent def start_engine(self): t = threading.Thread(target=self.game_loop, daemon=True) t.start() #
